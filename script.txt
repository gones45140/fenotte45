        function addBook() {
            const input = document.getElementById('new-bk-name');

        function updateBalance(name, val) {
            state.b[name] = parseFloat(val) || 0;

          function deleteBook(name) {
            if(confirm("Supprimer ce bookmaker ?")) {
                delete state.b[name];
                save();
            }

        function pari(isS) {
            let n, b, c, m, target = "", type = "", l = 1;
            const now = new Date();
            const time = now.toLocaleDateString('fr-FR', {day: '2-digit', month: '2-digit'}) + ' ' + now.toLocaleTimeString('fr-FR', {hour: '2-digit', minute:'2-digit'});
            
            if (isS) {
                const u = state.u.find(x => x.n === document.getElementById('c-unit').value);
                if (!u) return;
                n = u.n; target = document.getElementById('c-target').value || "-";
                type = document.getElementById('c-type').value || "-";
                b = document.getElementById('c-book').value;
                c = parseFloat(document.getElementById('c-cote').value) || 0;
                m = parseFloat(document.getElementById('c-mise').value) || 0;
                l = u.l;
            } else {
                target = document.getElementById('n-target').value || "Pari";
                type = document.getElementById('n-desc').value || "Simple";
                n = "PARI SIMPLE"; b = document.getElementById('n-book').value;
                c = parseFloat(document.getElementById('n-cote').value) || 0;
                m = parseFloat(document.getElementById('n-mise').value) || 0;
            }
            
            if (state.b[b] >= m && m > 0) {
                state.b[b] = (parseFloat(state.b[b]) - m).toFixed(2);
                state.h.unshift({ id: Date.now().toString(), n: n, target: target, type: type, b: b, l: l, m: m, cote: c, isS: isS, t: time });
                if (isS) { const u = state.u.find(x => x.n === n); if (u && u.l < 8) u.l++; }
                save();
            } else { alert("Solde insuffisant !"); }
        }

        function resolve(id, win) {
            const hIdx = state.h.findIndex(x => x.id === id);
            if(hIdx === -1) return;
            const h = state.h[hIdx];
            h.win = win; 
            if(win) {
                state.b[h.b] = (parseFloat(state.b[h.b]) + (h.m * h.cote)).toFixed(2);
                if(h.isS) { const u = state.u.find(x => x.n == h.n); if(u) u.l = 1; }
            }
            state.a.unshift(h);
            state.h.splice(hIdx, 1);
            save();
        }

        function save() { localStorage.setItem('gones_state', JSON.stringify(state)); render(); }
        function exportData() { document.getElementById('io-data').value = btoa(JSON.stringify(state)); }
        function importData() { try { state = JSON.parse(atob(document.getElementById('io-data').value)); save(); } catch(e) { alert("Erreur"); } }

        window.onload = () => {
            // Initialisation du state si vide
            if(!localStorage.getItem('gones_state')) {
                window.state = { start_bk: 0, b: {winamax: 0}, u: [], h: [], a: [] };
            } else {
                window.state = JSON.parse(localStorage.getItem('gones_state'));
            }
            render();
        };
